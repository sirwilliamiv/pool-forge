**Goal:** Land the shared types and seams that Wave 1's four tracks would otherwise collide on, with zero behaviour change.

**Architecture:** Three additive changes — a material-slot field and footprint resolver on the shape model, one function that owns pool area/perimeter, and a renderer registry that turns `SceneRoot` into a loop. Each is a pure refactor whose tests assert *identical* output to today.

**Tech Stack:** TypeScript strict, Vitest, three.js / react-three-fiber, Zustand.

## Global Constraints

See `2026-07-24-world-class-roadmap.md` → Global Constraints. Every task inherits them. Two that bite in this track specifically:

- `exactOptionalPropertyTypes` is on: never spread an object with a possibly-`undefined` field into a shape literal.
- 1 canvas unit = 1 inch. `poolFootprintMetrics` takes inches and returns feet / square feet.

**Drawings are stored as untyped JSON** in `Drawing.rootJson`, so adding fields to `Shape` needs **no Prisma migration**. Old drawings load with the new fields absent — every reader must treat them as optional.

```ts
// Which silhouette a pool is measured and drawn with. Points for the polygon
// case are normalised 0..1 inside the shape's bounding box, so a footprint is
// scale-independent and can be reused at any size.

import type { Point } from './polygon'
import type { DisplayHint } from '@/modules/editor/state/shapes'

export type Footprint =
  | { kind: 'rectangle' }
  | { kind: 'ellipse' }
  | { kind: 'polygon'; points: readonly Point[] }

export interface FootprintInput {
  displayHint?: DisplayHint | undefined
}

/**
 * Normalised silhouettes by StencilDef id. Wave 1 / T2 fills this in; an id
 * with no entry falls back to a rectangle so an unknown pool never measures
 * as zero area.
 */
export const FOOTPRINT_LIBRARY: Readonly<Record<string, readonly Point[]>> = {}

export function resolveFootprint(shape: FootprintInput): Footprint {
  const hint = shape.displayHint
  if (hint?.poolShape === 'ellipse') return { kind: 'ellipse' }
  if (hint?.poolShape === 'polygon') {
    const points = hint.footprintId ? FOOTPRINT_LIBRARY[hint.footprintId] : undefined
    if (points && points.length >= 3) return { kind: 'polygon', points }
  }
  return { kind: 'rectangle' }
}

/** Scale a normalised footprint into inches for a shape's bounding box. */
export function footprintPointsInInches(
  points: readonly Point[],
  widthInches: number,
  heightInches: number,
): Point[] {
  return points.map(([nx, ny]) => [nx * widthInches, ny * heightInches] as Point)
}
```

```ts
// The single owner of "how big is this pool". Every consumer — the measurement
// engine, the plan SVG, the 3D footprint mesh — asks here, so a new silhouette
// is added in one place and cannot disagree between the quote and the drawing.

import { footprintPointsInInches, resolveFootprint } from '@/lib/geometry/footprint'
import { ellipseAreaSqft, ellipsePerimeterLf } from '@/lib/geometry/ellipse'
import { polygonAreaSqft, polygonPerimeterLf } from '@/lib/geometry/polygon'
import { rectangleAreaSqft, rectanglePerimeterLf } from '@/lib/geometry/rectangle'
import type { DisplayHint } from '@/modules/editor/state/shapes'

export interface FootprintMetrics {
  areaSqft: number
  perimeterLf: number
}

export interface FootprintShape {
  width: number
  height: number
  displayHint?: DisplayHint | undefined
}

export function poolFootprintMetrics(shape: FootprintShape): FootprintMetrics {
  const footprint = resolveFootprint(shape)
  switch (footprint.kind) {
    case 'ellipse':
      return {
        areaSqft: ellipseAreaSqft(shape.width, shape.height),
        perimeterLf: ellipsePerimeterLf(shape.width, shape.height),
      }
    case 'polygon': {
      const points = footprintPointsInInches(footprint.points, shape.width, shape.height)
      return { areaSqft: polygonAreaSqft(points), perimeterLf: polygonPerimeterLf(points) }
    }
    case 'rectangle':
      return {
        areaSqft: rectangleAreaSqft(shape.width, shape.height),
        perimeterLf: rectanglePerimeterLf(shape.width, shape.height),
      }
  }
}
```

```tsx
'use client'

import { ShapeKind, type Shape } from '@/modules/editor/state/shapes'
import { feet } from '@/lib/three/units'

import { Bubblers } from './objects/Bubblers'
import { Coping } from './objects/Coping'
import { Deck } from './objects/Deck'
import { EllipsePool } from './objects/EllipsePool'
import { GenericStencil } from './objects/GenericStencil'
import { LedLights } from './objects/LedLights'
import { PoolWalls } from './objects/PoolWalls'
import { Spa } from './objects/Spa'
import { Spillover } from './objects/Spillover'
import { Steps } from './objects/Steps'
import { SunShelf } from './objects/SunShelf'
import { TileBand } from './objects/TileBand'
import { Water } from './objects/Water'

export type ShapeRenderer = (props: { shape: Shape }) => React.ReactNode

const _stencilRenderers = new Map<string, ShapeRenderer>()

export const STENCIL_RENDERERS: ReadonlyMap<string, ShapeRenderer> = _stencilRenderers

/** Later-registered renderers win, so a track can override a placeholder. */
export function registerStencilRenderer(stencilId: string, render: ShapeRenderer): void {
  _stencilRenderers.set(stencilId, render)
}

// Object components other than GenericStencil draw at the scene origin, so a
// registered stencil renderer is wrapped in a positioned group.
function positioned(render: ShapeRenderer): ShapeRenderer {
  return function Positioned({ shape }) {
    const cx = feet(shape.x + shape.width / 2)
    const cz = feet(shape.y + shape.height / 2)
    return (
      <group position={[cx, 0, cz]}>{render({ shape })}</group>
    )
  }
}

for (const id of [
  'pool.standard-steps',
  'pool.one-step',
  'pool.step-sets',
  'pool.corner-steps',
  'pool.square-steps',
]) {
  registerStencilRenderer(id, positioned(({ shape }) => <Steps shape={shape} />))
}
for (const id of ['feature.bubblers', 'feature.deck-jets']) {
  registerStencilRenderer(id, positioned(({ shape }) => <Bubblers shape={shape} />))
}
registerStencilRenderer('feature.light', positioned(({ shape }) => <LedLights shape={shape} />))
for (const id of ['feature.sun-shelf', 'feature.tanning-ledge']) {
  registerStencilRenderer(id, positioned(({ shape }) => <SunShelf shape={shape} />))
}
registerStencilRenderer('pool.spa', positioned(({ shape }) => <Spa shape={shape} />))

const RectanglePoolRenderer: ShapeRenderer = ({ shape }) =>
  shape.displayHint?.poolShape === 'ellipse' ? (
    <EllipsePool shape={shape} />
  ) : (
    <>
      <PoolWalls shape={shape} />
      <Water shape={shape} />
      <Coping shape={shape} />
      <TileBand shape={shape} />
    </>
  )

const GenericStencilRenderer: ShapeRenderer = ({ shape }) => <GenericStencil shape={shape} />

const BY_KIND: Record<ShapeKind, ShapeRenderer> = {
  [ShapeKind.RECTANGLE_POOL]: RectanglePoolRenderer,
  [ShapeKind.SUN_SHELF]: ({ shape }) => <SunShelf shape={shape} />,
  [ShapeKind.BENCH]: ({ shape }) => <SunShelf shape={shape} />,
  [ShapeKind.SPA]: ({ shape }) => (
    <>
      <Spa shape={shape} />
      <Spillover shape={shape} />
    </>
  ),
  [ShapeKind.CONCRETE_DECK]: ({ shape }) => <Deck shape={shape} />,
  [ShapeKind.PAVER_DECK]: ({ shape }) => <Deck shape={shape} />,
  [ShapeKind.GRASS_AREA]: ({ shape }) => <Deck shape={shape} />,
  [ShapeKind.STENCIL]: GenericStencilRenderer,
}

export function getShapeRenderer(shape: Shape): ShapeRenderer {
  if (shape.kind === ShapeKind.STENCIL) {
    return _stencilRenderers.get(shape.stencilId) ?? GenericStencilRenderer
  }
  return BY_KIND[shape.kind]
}

'use client'

import { useShapesStore } from '@/modules/editor/state/shapesStore'
import { usePresentationFlags } from '@/modules/editor/state/viewStore'

import { BuildOverlay } from './BuildOverlay'
import { EquipmentPad } from './objects/EquipmentPad'
import { HouseWall } from './objects/HouseWall'
import { Loungers } from './objects/Loungers'
import { PlanOverlay } from './PlanOverlay'
import { Trees } from './objects/Trees'
import { getShapeRenderer } from './renderers'

export function SceneRoot() {
  const shapes = useShapesStore((s) => s.shapes)
  const flags = usePresentationFlags()

  return (
    <group name="scene-root">
      {shapes.map((shape) => {
        if (shape.hidden) return null
        const Renderer = getShapeRenderer(shape)
        return (
          <group key={`${shape.id}-${shape.width}-${shape.height}-${shape.rotation ?? 0}`}>
            <Renderer shape={shape} />
          </group>
        )
      })}

      {flags.showSiteContext ? (
        <>
          <HouseWall position={[0, 7, -28]} />
          <Trees />
          <Loungers />
        </>
      ) : null}

      {flags.showEquipmentPad ? <EquipmentPad position={[24, 0, -22]} /> : null}

      {flags.showPlanOverlay ? <PlanOverlay /> : null}
      {flags.showConstructionOverlay ? <BuildOverlay /> : null}
    </group>
  )
}
```
