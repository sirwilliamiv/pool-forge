# Wave 1 / Track 3 — Pricing Covers the Catalog

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every price-book category prices from a real measurement or is honestly marked manual, formulas in the price book actually evaluate, and screen enclosures — the beachhead — price from cage surface rather than deck footprint.

**Architecture:** Three independent pieces: a small safe formula evaluator over a fixed variable set; measurement additions for the categories that currently return 0; and a cage-area model for screen. All three land in the pure pricing/measurement layer, so they are fully unit-testable with no DB.

**Tech Stack:** TypeScript, Zod, Vitest.

## Global Constraints

See `2026-07-24-world-class-roadmap.md` → Global Constraints. Two specific to this track:

- **Never `eval`.** The formula evaluator parses a fixed grammar. A price book is user-editable data; running it as JavaScript is a code-execution hole.
- Quantities round to 2 decimals **before** the line total is derived, so `quantity × unitPrice` always equals the printed total. `computeQuote` already does this — keep it.

Create `src/modules/pricing/formula.ts`:

```ts
// A deliberately tiny expression language for price-book formulas.
//
// Grammar: number | identifier | + - * / | ( ) | unary minus.
// Identifiers resolve only against the variable map passed in — see
// `variables.ts` for what a formula is allowed to see. NEVER use eval() or
// `new Function()`: a price book is user-editable data and this runs on the
// server, so anything outside this grammar must be a parse error.

export type FormulaResult = { ok: true; value: number } | { ok: false; error: string }

type Token = { kind: 'number'; value: number } | { kind: 'name'; value: string } | { kind: 'op'; value: string }

const PRECEDENCE: Readonly<Record<string, number>> = { '+': 1, '-': 1, '*': 2, '/': 2 }

function tokenize(expression: string): Token[] | string {
  const tokens: Token[] = []
  const pattern = /\s*(\d+\.?\d*|[A-Za-z_][A-Za-z0-9_]*|[+\-*/()])/y
  let index = 0
  while (index < expression.length) {
    pattern.lastIndex = index
    const match = pattern.exec(expression)
    if (!match) {
      const rest = expression.slice(index).trim()
      if (rest === '') break
      return `unexpected character: ${rest[0]}`
    }
    const raw = match[1]!
    if (/^\d/.test(raw)) tokens.push({ kind: 'number', value: Number(raw) })
    else if (/^[A-Za-z_]/.test(raw)) tokens.push({ kind: 'name', value: raw })
    else tokens.push({ kind: 'op', value: raw })
    index = pattern.lastIndex
  }
  return tokens
}

export function evaluateFormula(
  expression: string,
  variables: Readonly<Record<string, number>>,
): FormulaResult {
  const tokens = tokenize(expression)
  if (typeof tokens === 'string') return { ok: false, error: tokens }
  if (tokens.length === 0) return { ok: false, error: 'empty expression' }

  const values: number[] = []
  const operators: string[] = []
  let expectOperand = true

  function applyTop(): string | null {
    const op = operators.pop()
    if (!op || op === '(') return 'malformed expression'
    const right = values.pop()
    const left = values.pop()
    if (right === undefined || left === undefined) return 'malformed expression'
    if (op === '/' && right === 0) return 'division by zero'
    values.push(op === '+' ? left + right : op === '-' ? left - right : op === '*' ? left * right : left / right)
    return null
  }

  for (const token of tokens) {
    if (token.kind === 'number') {
      if (!expectOperand) return { ok: false, error: 'malformed expression' }
      values.push(token.value)
      expectOperand = false
      continue
    }
    if (token.kind === 'name') {
      if (!expectOperand) return { ok: false, error: 'malformed expression' }
      const value = variables[token.value]
      if (typeof value !== 'number') return { ok: false, error: `unknown variable: ${token.value}` }
      values.push(value)
      expectOperand = false
      continue
    }
    if (token.value === '(') {
      if (!expectOperand) return { ok: false, error: 'malformed expression' }
      operators.push('(')
      continue
    }
    if (token.value === ')') {
      if (expectOperand) return { ok: false, error: 'malformed expression' }
      while (operators.length > 0 && operators[operators.length - 1] !== '(') {
        const err = applyTop()
        if (err) return { ok: false, error: err }
      }
      if (operators.pop() !== '(') return { ok: false, error: 'unbalanced parentheses' }
      expectOperand = false
      continue
    }
    // Unary minus in operand position: push a zero so `-2.5` parses as `0 - 2.5`.
    if (expectOperand) {
      if (token.value !== '-') return { ok: false, error: 'malformed expression' }
      values.push(0)
    }
    const precedence = PRECEDENCE[token.value]
    if (precedence === undefined) return { ok: false, error: `unsupported operator: ${token.value}` }
    while (operators.length > 0) {
      const top = operators[operators.length - 1]!
      if (top === '(' || (PRECEDENCE[top] ?? 0) < precedence) break
      const err = applyTop()
      if (err) return { ok: false, error: err }
    }
    operators.push(token.value)
    expectOperand = true
  }

  if (expectOperand) return { ok: false, error: 'malformed expression' }
  while (operators.length > 0) {
    if (operators[operators.length - 1] === '(') return { ok: false, error: 'unbalanced parentheses' }
    const err = applyTop()
    if (err) return { ok: false, error: err }
  }
  const result = values.pop()
  if (result === undefined || values.length > 0) return { ok: false, error: 'malformed expression' }
  if (!Number.isFinite(result)) return { ok: false, error: 'result is not a finite number' }
  return { ok: true, value: result }
}

export function parseFormulaColumn(json: unknown): string | null {
  if (typeof json === 'string') return json.trim() === '' ? null : json
  if (json && typeof json === 'object' && !Array.isArray(json)) {
    const expr = (json as { expression?: unknown }).expression
    if (typeof expr === 'string' && expr.trim() !== '') return expr
  }
  return null
}
```
